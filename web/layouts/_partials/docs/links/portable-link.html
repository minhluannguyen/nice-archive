{{- $destination := .Destination -}} {{- $url := urls.Parse .Destination -}} {{- if and (not $url.IsAbs) (not (strings.HasPrefix .Destination "#")) -}} {{- $path := strings.TrimPrefix "./" $url.Path -}} {{- $pagePath := strings.TrimSuffix ".md" $path -}} {{- $pagePath = strings.TrimSuffix "/_index" $pagePath -}} {{- with (.Page.GetPage $pagePath) -}} {{- $destination = .RelPermalink -}} {{- else with (.Page.Resources.Get $pagePath) -}} {{- $destination = .RelPermalink -}} {{- else with (resources.Get $pagePath) -}} {{- $destination = .RelPermalink -}} {{- else -}} {{- $sourcePath := path.Clean (path.Join (path.Dir .Page.File.Filename) $url.Path) -}} {{- $repoRoot := path.Clean (path.Join hugo.WorkingDir "..") -}} {{- $insideRepo := strings.HasPrefix $sourcePath (printf "%s/" $repoRoot) -}} {{- $repoPath := strings.TrimPrefix (printf "%s/" $repoRoot) $sourcePath -}} {{- if $insideRepo -}} {{- $kind := cond (strings.HasSuffix $url.Path "/") "tree" "blob" -}} {{- $destination = printf "%s/%s/main/%s" (strings.TrimSuffix "/" .Page.Site.Params.BookRepo) $kind $repoPath -}} {{- else -}} {{- warnf "Reference '%s' not found in '%s'" .Destination .Page.RelPermalink -}} {{- end -}} {{- end -}} {{- with $url.RawQuery -}} {{- $destination = print $destination "?" . -}} {{- end -}} {{- with $url.Fragment -}} {{- $destination = print $destination "#" . -}} {{- end -}} {{- end -}} {{- return $destination -}}
